/**
 * Builds default values and opts for the model
 * @param {*} opts 
 * @returns default values and opts
 * @memberof newModelParent
 * @access private
 */
const modelDefaults = () => {
    const defaultValues = {
        constants: {
            interfaceThickness: standards.INTERFACE_THICKNESS,
        },
        opts: {},
        dims: {
            size: [
                math.inchesToMm(2),
                math.inchesToMm(4),
                math.inchesToMm(1),
            ],
        },
        points: {},
        components: {},
        types: {
            default: { id: 'default', desc: 'Default model' },
            alt: { id: 'alt', desc: 'Alternate model' },
        },
    }

    const defaultOpts = {
        fitGap: standards.FIT_GAP,
        size: defaultValues.dims.size,
        type: defaultValues.types.default.id,
        interfaceThickness: defaultValues.constants.interfaceThickness,
    }

    return {
        opts: defaultOpts,
        vals: defaultValues,
    }
}


//------------------------------------------------------------------------------


/**
 * Initializes options with user input
 * @param {*} opts 
 * @returns model properties
 * @memberof newModelParent
 * @access private
 */
const modelOpts = (opts) => {
    const defaults = modelDefaults()
    console.log('modelOpts() -- opts', opts)

    // User options
    const {
        size = defaults.opts.size,
        type = defaults.opts.type,
        interfaceThickness = defaults.opts.interfaceThickness,
        fitGap = defaults.opts.fitGap,
    } = opts

    const initOpts = {
        size,
        type,
        interfaceThickness,
        fitGap,
    }

    console.log('modelOpts() -- initOpts', initOpts)

    return initOpts
}


//------------------------------------------------------------------------------


/**
 * Builds model properties from the given opts
 * @param {*} opts 
 * @returns model properties
 * @memberof newModelParent
 * @access private
 */
const modelProps = (opts) => {
    const defaults = modelDefaults()
    console.log('modelProps() -- prop opts', opts)

    const {
        size,
        type,
        interfaceThickness,
        fitGap,
    } = opts

    /* ----------------------------------------
    * Prop calculations
    * ------------------------------------- */

    const width = size[0]
    const depth = size[1]
    const height = size[2]

    /* ----------------------------------------
    * Preparing Model Properties, Dimensions
    * ------------------------------------- */

    /** Constant values for model */
    const modelConstants = defaults.vals.constants

    /** Derived user options for the model */
    const modelOpts = {
        type,
    }

    /** Various dimensions for model */
    const modelDims = {
        size,
        width,
        depth,
        height,
        interfaceThickness,
        fitGap,
    }

    /** Various key points for model */
    const modelPoints = defaults.vals.points

    /** Components used by model */
    const modelComponents = defaults.vals.components

    /* ---------------------------------------------
    *  Model Properties
    * ----------------------------------------------
    * Properties accessible to all model functions.
    * --------------------------------------------- */

    const modelProperties = {
        constants: modelConstants,
        opts: modelOpts,
        dims: modelDims,
        points: modelPoints,
        components: modelComponents,
    }

    console.log('modelProps() -- modelProperties', modelProperties)

    return modelProperties
}


//------------------------------------------------------------------------------


/**
 * New Model
 * @param {*} opts 
 * @returns Array with model, parts, and properties: [`geom3`, `Object.<string, geom3>`, `Object.<string, any>`]
 * @memberof newModelParent
 */
const model = (opts) => {
    const defaults = modelDefaults()
    const initOpts = modelOpts(opts)
    const modelProperties = modelProps(initOpts)

    /* ----------------------------------------
     * Modelling, Component/Assembly Modules
     * ------------------------------------- */

    /** Sub-component 01 */
    const subcomponent1 = (modelProps) => {
        const {
            opts,
            dims,
            points,
        } = modelProps

        return sphere()
    }

    /** Sub-component 02 */
    const subcomponent2 = (modelProps) => {
        const {
            opts,
            dims,
            points,
        } = modelProps

        return cube()
    }

    /* ----------------------------------------
     * Complete Assembly
     * ------------------------------------- */

    /** Final Assembly */
    const finalAssembly = (modelProps) => {
        let subComp1 = subcomponent1(modelProps)
        let subComp2 = subcomponent2(modelProps)

        return union(
            subComp1,
            subComp2,
        )
    }

    /* ----------------------------------------
     * Outputs
     * ------------------------------------- */

    let subComp1 = subcomponent1(modelProperties)
    let subComp2 = subcomponent2(modelProperties)

    let mainModel = finalAssembly(modelProperties)

    let modelParts = {
        subcomponent1: subComp1,
        subcomponent2: subComp2,
    }

    return [mainModel, modelParts, modelProperties]
}

/* ----------------------------------------
 * Model / Scene Prep

const outer = subtract(
    cube({ size: 10 }),
    sphere({ radius: 6.8 })
)

const inner = intersect(
    sphere({ radius: 4 }),
    cube({ size: 7 })
)

/**
 * Main function to be rendered by JSCAD
 * @returns `geom3` or `geom3[]`
 * @ignore
 */
function main() {
    const spaceUnit = math.inchesToMm(3.5)

    return [
        translate([spaceUnit * 0, spaceUnit * 0, spaceUnit * 0], inner),
        translate([spaceUnit * 0, spaceUnit * 0, spaceUnit * 0], outer),
    ]
}
